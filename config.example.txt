# -- Custom messages -----------------------------------------------------------
# Named message slots - define as many as you like. Names may use letters,
# digits, _ and - (the /messages web page enforces this for slots edited there).
# A slot is active when its _text key exists and today falls within [start, end].
#
# How often to cycle through active messages (seconds, 10-3600). Default: 60
msg_interval=60
#
# Slot format:
#   message_<name>_text      - text to scroll (required to activate the slot)
#   message_<name>_start     - YYYY-MM-DD - hide before this date (optional)
#   message_<name>_end       - YYYY-MM-DD - hide after this date (optional)
#   message_<name>_countdown - YYYY-MM-DD - appends ": 45d" ("today!" on the
#                              day, "+Nd" after) unless the text contains {}
#
# {} in the text is replaced by the countdown number.
# A countdown hides itself from the day after its target unless _end is set later.
# {key} placeholders insert RuntimeStore values (temp_c, weather_id, ...),
# DeviceStore values (ip, hostname, heap, uptime, ...) or DataStore config keys.
# Example: always-on greeting
message_1_text=Hello, world!

# Example: date-bounded announcement
